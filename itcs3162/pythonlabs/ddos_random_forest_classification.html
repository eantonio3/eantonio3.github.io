# Remove duplicate rows to ensure only unique traffic flows remain
df = df.drop_duplicates()
# Drop rows with missing values so the model isn't affected by incomplete records
df = df.dropna()
# Standardize column names (remove spaces & lowercase)
df.columns = df.columns.str.strip().str.lower().str.replace(" ", "_")
# Drop non-numeric identifier columns that don't help classification
df = df.drop(["flow_id", "source_ip", "destination_ip", "timestamp", "fwd_avg_bulk_rate", "fwd_urg_flags", "ece_flag_count", "bwd_urg_flags", "fwd_avg_bytes/bulk", "bwd_avg_bytes/bulk", "bwd_avg_bulk_rate", "bwd_psh_flags", "cwe_flag_count", "fwd_avg_packets/bulk", "bwd_avg_packets/bulk"], axis=1)

# Encode categorical target labels into numeric classes (BENIGN=0, DDoS=1)
label_encoder = LabelEncoder()
df["label"] = label_encoder.fit_transform(df["label"])

df.replace([np.inf, -np.inf], np.nan, inplace=True)
df = df.dropna()

# Get predicted probabilities for the positive class (1 = attack, 0 = benign)
y_prob = rf.predict_proba(X_test)[:, 1]

# Compute ROC curve and AUC
fpr, tpr, thresholds = roc_curve(y_test, y_prob)
roc_auc = roc_auc_score(y_test, y_prob)

# Plot ROC curve
plt.figure(figsize=(8,6))
